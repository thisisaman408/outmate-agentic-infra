import ipinfo
from app.services.explorium import explorium_client  # Existing

class VisitorEnricher:
    def __init__(self):
        self.ipinfo = ipinfo.Client(settings.IPINFO_TOKEN)
    
    async def enrich_ip(self, ip: str, url: str, intent_score: float):
        # Free IPinfo company lookup
        company_data = self.ipinfo.org(ip)
        resolution = {
            "company": company_data.org,
            "domain": company_data.hostname,
            "geo": company_data.country,
            "confidence": 0.6
        }
        
        # High-intent: Enrich.so person lookup
        if intent_score > 0.7 or "/pricing" in url or "/demo" in url:
            enrich_response = await self._enrich_so_lookup(ip)
            if enrich_response.get("data", {}).get("email"):
                # Explorium verification (existing service)
                explorium_data = await explorium_client.enrich({
                    "email": enrich_response["data"]["email"]
                })
                resolution.update({
                    "person": enrich_response["data"],
                    "explorium": explorium_data,
                    "confidence": 0.85,
                    "intent_score": intent_score
                })
        
        return resolution
    
    async def _enrich_so_lookup(self, ip: str):
        # Enrich.so API call
        response = await httpx.AsyncClient().post(
            "https://api.enrich.so/ip-to-company",
            json={"ip": ip},
            headers={"Authorization": f"Bearer {settings.ENRICH_API_KEY}"}
        )
        return response.json()

@router.post("/track")
async def track_visitor(
    url: str = Form(...),
    referrer: str = Form(None),
    user_agent: str = Header(...),
    x_forwarded_for: str = Header(None),
    pixel_key: str = Header(..., alias="X-Pixel-Key")
):
    org_id = crud.get_org_by_pixel_key(pixel_key)
    if not org_id:
        raise HTTPException(401, "Invalid pixel key")
    
    ip = x_forwarded_for.split(",").strip() if x_forwarded_for else "127.0.0.1"
    intent_score = 1.0 if any(x in url.lower() for x in ["/pricing", "/demo", "/contact"]) else 0.5
    
    # Redis deduplication (1h per IP/domain)
    dedupe_key = f"visits:{org_id}:{ip}:{urlparse(url).netloc}"
    if redis_client.get(dedupe_key):
        return {"status": "deduplicated"}
    redis_client.setex(dedupe_key, 3600, "1")
    
    # Queue processing
    task = process_visitor.delay(org_id, {
        "ip": ip, "url": url, "referrer": referrer, 
        "user_agent": user_agent, "intent_score": intent_score
    })
    return {"status": "queued", "task_id": task.id}

@celery.task(bind=True, max_retries=3)
def process_visitor(self, org_id: str, data: dict):
    try:
        enricher = VisitorEnricher()
        resolution = enricher.enrich_ip(data["ip"], data["url"], data["intent_score"])
        
        visit = crud.create_visit(org_id, {**data, "resolution": resolution})
        matched = resolution["confidence"] > 0.7
        
        if matched:
            crud.update_visit(visit.id, {"matched": True})
            await send_webhooks(org_id, resolution, visit.id)
            
    except Exception as exc:
        logger.error(f"Visitor processing failed: {exc}")
        raise self.retry(countdown=60)
🎨 FRONTEND IMPLEMENTATION (PHASE 3)
1. Pixel JS (public/pixel.js)
javascript
(function(w,d,k){
    w._outmateVisitor={key:k,track:function(){
        const xhr=new XMLHttpRequest();
        xhr.open('POST','/api/track',true);
        xhr.setRequestHeader('Content-Type','application/json');
        xhr.setRequestHeader('X-Pixel-Key',k);
        xhr.send(JSON.stringify({
            url:location.href,
            referrer:document.referrer,
            userAgent:navigator.userAgent
        }));
    }};
    const s=d.createElement('script');
    s.src=`/pixel.js?key=${k}`;
    d.head.appendChild(s);
    w.addEventListener('load',w._outmateVisitor.track);
    w.addEventListener('pagehide',w._outmateVisitor.track);
})(window,document,'YOUR_PIXEL_KEY');
